cmake_minimum_required(VERSION 3.20)
project(FlexDrive VERSION 0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(raylib)
add_executable(flex_drive src/main.cpp)
target_link_libraries(flex_drive PRIVATE raylib)
if(MSVC)
 target_compile_options(flex_drive PRIVATE /W4)
else()
 target_compile_options(flex_drive PRIVATE -Wall -Wextra -Wpedantic)
endif()
